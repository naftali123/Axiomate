function vStatement(id){

  var names=['נקרא מהטקסט','הוצע כטענה','עבר בדיקה','הושווה לתיעוד'];

function route(){
  var h=(location.hash||'#list').slice(1),p=h.split('-'),k=p[0],rest=p.slice(1).join('-');
  if(k==='entity')vEntity(rest);else if(k==='statement')vStatement(rest);else if(k==='about')vAbout();else vList();
  document.querySelectorAll('nav a').forEach(function(a){var on=a.getAttribute('data-v')===(k==='about'?'about':'list');if(on)a.setAttribute('aria-current','page');else a.removeAttribute('aria-current')});
  window.scrollTo(0,0);
}
window.addEventListener('hashchange',route);route();
