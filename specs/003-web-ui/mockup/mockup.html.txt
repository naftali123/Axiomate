/* reader linking + filters */
var ev=document.getElementById('ev'),cl=document.getElementById('cls');
function hot(id,on){
  ev.querySelectorAll('[data-c="'+id+'"]').forEach(function(e){e.classList.toggle('hot',on)});
  cl.querySelectorAll('[data-c="'+id+'"]').forEach(function(e){e.classList.toggle('hot',on)});
}
document.querySelectorAll('[data-c]').forEach(function(e){
  var id=e.getAttribute('data-c');
  e.addEventListener('mouseenter',function(){hot(id,true)});e.addEventListener('mouseleave',function(){hot(id,false)});
  e.addEventListener('focus',function(){hot(id,true)});e.addEventListener('blur',function(){hot(id,false)});
  e.addEventListener('click',function(){var o=e.tagName==='BUTTON'?ev.querySelector('[data-c="'+id+'"]'):cl.querySelector('[data-c="'+id+'"]');if(o&&o.scrollIntoView)o.scrollIntoView({block:'nearest'})});
});
document.getElementById('filters').addEventListener('click',function(e){
  var b=e.target.closest('button');if(!b)return;
  this.querySelectorAll('button').forEach(function(x){x.setAttribute('aria-pressed',String(x===b))});
  var f=b.getAttribute('data-f');
  cl.querySelectorAll('.cl').forEach(function(c){c.hidden=!(f==='all'||c.getAttribute('data-k').split(' ').indexOf(f)>-1)});
});

/* routing */
var VIEWS=['document','finding','trust'];
function route(){
  var h=(location.hash||'').slice(1);if(VIEWS.indexOf(h)<0)h='document';
  VIEWS.forEach(function(v){document.getElementById('v-'+v).classList.toggle('on',v===h)});
  document.querySelectorAll('nav a').forEach(function(a){if(a.getAttribute('data-v')===h)a.setAttribute('aria-current','page');else a.removeAttribute('aria-current')});
}
window.addEventListener('hashchange',route);route();
